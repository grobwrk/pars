163 std::span<uint8_t> span() const
164 {
165 return std::visit(
166 [](auto&& b) {
167 using T = std::decay_t<decltype(b)>;
168
169 if constexpr (std::is_same_v<
170 T, std::shared_ptr<flatbuffers::DetachedBuffer>>)
171 {
172 auto buf =
173 *static_cast<const std::shared_ptr<flatbuffers::DetachedBuffer>*>(
174 &b);
175
176 return std::span<uint8_t>(buf->begin(),
177 std::distance(buf->begin(), buf->end()));
178 }
179 else if constexpr (std::is_same_v<T, std::shared_ptr<nngxx::msg>>)
180 {
181 auto msg = *static_cast<const std::shared_ptr<nngxx::msg>*>(&b);
182
183 return std::span<uint8_t>(msg->body().data<uint8_t>() + 8,
184 msg->body().size() - 8);
185 }
186 else
187 static_assert(!std::is_same_v<T, T>, "non-exhaustive visitor!");
188 },
189 buf_m);
190 }

203struct serialize
204{
205 template<event_c event_t>
206 static nngxx::msg to_network(event_t& ev)
207 {
208 // 1. get the serialized data
209 std::span<uint8_t> span = ev.span();
210
211 // 2. compute the event_hash
212 auto event_hash = uuid<klass<event_t>>::hash;
213
214 // 3. create the nngxx::msg to hold the hash+event
215 auto m = nngxx::make_msg(sizeof(event_hash) + span.size()).value_or_abort();
216 auto b = m.body();
217
218 // 4. append the event hash
219 std::memcpy(b.data<char>(), &event_hash, sizeof(event_hash));
220
221 // 5. append the serialized event
222 memcpy(b.data<uint8_t>() + sizeof(event_hash), span.data(), span.size());
223
224 pars::debug(SL, lf::event, "Serialized Event [{}] to Message [{}]", ev, m);
225
226 return m;
227 }
228
229 template<event_c event_t>
230 static event_t to_event(const nngxx::msg& m)
231 {
232 // 1. compute received and requested event hash
233 auto recv_event_hash = net::hash_from_msg(m);
234 auto req_event_hash = uuid<klass<event_t>>::hash;
235
236 // 2. check they correspond
237 if (recv_event_hash - req_event_hash != 0)
238 throw std::runtime_error("Requested event mismatch!");
239
240 // 3. deserialize event
241 auto ev = obb<event_t>::from(std::move(m));
242
243 pars::debug(SL, lf::event, "Deserialized Message [{}] to Event [{}]", m,
244 ev);
245
246 return ev;
247 }
248};
