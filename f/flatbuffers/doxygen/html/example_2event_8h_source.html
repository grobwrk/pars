52// a client send this to request a fib_n computation
53struct fib_requested : pars::ev::fb<FibRequestedBuilder, fib_requested>
54{
55 static auto make(std::size_t work_id, uint64_t n, bool use_fast_fib)
56 {
57 return pars::ev::obb<fib_requested>::using_size(64)
58 .and_then(&fib_requested::builder::add_work_id, work_id)
59 .and_then(&fib_requested::builder::add_n, n)
60 .and_then(&fib_requested::builder::add_use_fast_fib, use_fast_fib)
61 .build();
62 }
63
64 auto format_to(std::format_context& ctx) const -> decltype(ctx.out())
65 {
66 auto t = table();
67
68 return std::format_to(ctx.out(), "fib_requested({},{},{})", t->work_id(),
69 t->n(), t->use_fast_fib() ? "fast_fib" : "slow_fib");
70 }
71};
