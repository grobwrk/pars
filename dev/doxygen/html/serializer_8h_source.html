124 std::span<uint8_t> span() const
125 {
126 return std::visit(
127 [](auto&& b) {
128 using T = std::decay_t<decltype(b)>;
129
130 if constexpr (std::is_same_v<
131 T, std::shared_ptr<flatbuffers::DetachedBuffer>>)
132 {
133 auto buf =
134 *static_cast<const std::shared_ptr<flatbuffers::DetachedBuffer>*>(
135 &b);
136
137 return std::span<uint8_t>(buf->begin(),
138 std::distance(buf->begin(), buf->end()));
139 }
140 else if constexpr (std::is_same_v<T, std::shared_ptr<nngxx::msg>>)
141 {
142 auto msg = *static_cast<const std::shared_ptr<nngxx::msg>*>(&b);
143
144 return std::span<uint8_t>(msg->body().data<uint8_t>() + 8,
145 msg->body().size() - 8);
146 }
147 else
148 static_assert(!std::is_same_v<T, T>, "non-exhaustive visitor!");
149 },
150 buf_m);
151 }
152
153 table_type const* table() const
154 {
155 auto table = flatbuffers::GetRoot<table_type>(span().data());
156
157 return table;
158 }

171struct serialize
172{
173 template<event_c event_t>
174 static nngxx::msg to_network(event_t& ev)
175 {
176 // 1. get the serialized data
177 std::span<uint8_t> span = ev.span();
178
179 // 2. compute the event_hash
180 auto event_hash = uuid<klass<event_t>>::hash;
181
182 // 3. create the nngxx::msg to hold the hash+event
183 auto m = nngxx::make_msg(sizeof(event_hash) + span.size()).value_or_abort();
184 auto b = m.body();
185
186 // 4. append the event hash
187 std::memcpy(b.data<char>(), &event_hash, sizeof(event_hash));
188
189 // 5. append the serialized event
190 memcpy(b.data<uint8_t>() + sizeof(event_hash), span.data(), span.size());
191
192 pars::debug(SL, lf::event, "Serialized Event [{}] to Message [{}]", ev, m);
193
194 return m;
195 }
196
197 template<event_c event_t>
198 static event_t to_event(const nngxx::msg& m)
199 {
200 // 1. compute received and requested event hash
201 auto recv_event_hash = hash_from_msg(m);
202 auto req_event_hash = uuid<klass<event_t>>::hash;
203
204 // 2. check they correspond
205 if (recv_event_hash - req_event_hash != 0)
206 throw std::runtime_error("Requested event mismatch!");
207
208 // 3. deserialize event
209 auto ev = builder_for<event_t>::from(std::move(m));
210
211 pars::debug(SL, lf::event, "Deserialized Message [{}] to Event [{}]", m,
212 ev);
213
214 return ev;
215 }
216};
